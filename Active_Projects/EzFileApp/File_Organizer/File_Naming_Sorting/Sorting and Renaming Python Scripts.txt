import os
import argparse

def validate_input(filename, initial_code):
    if not filename:
        raise ValueError("Invalid filename. Please enter a non-empty string.")
    if not initial_code.strip():
        raise ValueError("Initial code cannot be empty or whitespace.")

def create_file(file_path, initial_code, overwrite):
    if not overwrite and os.path.exists(file_path):
        print(f"Error: File '{file_path}' already exists. Use --overwrite flag to delete it and create a new file.")
        return

    with open(file_path, 'w') as file:
        file.write(initial_code)

if __name__ == '__main__':
    # Parse user input for filename, initial_code, and overwrite flag
    parser = argparse.ArgumentParser(description='Create a new Python file')
    parser.add_argument('filename', help='The name of the new Python file')
    parser.add_argument('--initial_code', help='Initial code to put in the new file', default='')
    parser.add_argument('--overwrite', action='store_true', help='Overwrite the file if it exists')

    args = parser.parse_args()
    try:
        validate_input(args.filename, args.initial_code)
        create_file(args.filename, args.initial_code, args.overwrite)
    except ValueError as e:
        print(e)
    else:
        print(f"Successfully created {args.filename} with initial code:\n{args.initial_code}")
